#include <stdio.h>
struct Book
{
    int id;
    char name[50];
    char author[50];
    float price;
};
int main()
{
    struct Book b;
    printf("Enter the book id:");
    scanf("%d",&b.id);
    printf("Enter the bookname:");
    scanf("%s",b.name);
    printf("enter the authorname:");
    scanf("%s",b.author);
    printf("Enter the price:");
    scanf("%f",&b.price);
    printf("Book id:%d\n",b.id);
    printf("Book name:%s\n",b.name);
    printf("Book author:%s\n",b.author);
    printf("Book price:%.2f\n",b.price);
    return 0;
}
