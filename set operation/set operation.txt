 1 #include<stdio.h>
  2 #define SIZE 9

  5 void readset(bit a[],int n);
  6 void printset(const bit a[]);

  9 void difference(const bit a[], const bit b[],bit c[]);

 37 void readset(bit a[],int n)
 38 {
 39     int i,k,x;
 40     if(n<0||n>SIZE)
 41     {
 42         printf("invalid size;must be between 0 and %d\n",SIZE);
 43         return;
 44     }
 45     printf("Enter %d elements:",n);
 46     for(i=0;i<n;i++)
 47     {
 48         if(scanf("%d",&x)!=1)return;
 49         for(k=0;k<SIZE;k++)
 50         {
 51             if(uset[k]==x)
 52             {
 53                 a[k].x=1;break;
 54             }
 55         }
 56         if(k==SIZE)
 57             printf("%d is not in the universal set,ignored\n",x);
 58     }
 59 }
 60 void printset(const bit a[])
 61 {
 62     int k,first=1;
 63     printf("{");
 64     for(k=0;k<SIZE;k++)
 65     {
 66         if(a[k].x)
 67         {
 68             printf(first ?"%d":"%d",uset[k]);
 69             first=0;
 70         }
 71     }
 72     printf("}\n");
 73 }
 74 void unionset(const bit a[],const bit b[],bit c[])
 75 {
 76     for(int k=0;k<SIZE;k++)
 77         c[k].x=a[k].x | b[k].x;
 78 }
 79 void intersect(const bit a[],const bit b[],bit c[])
 80 {
 81     for(int k=0;k<SIZE;k++)
 82         c[k].x=a[k].x & b[k].x;
 83 }
 84 void difference(const bit a[],const bit b[],bit c[])
 85 {
 86     for(int k=0;k<SIZE;k++)
 87         c[k].x=a[k].x && !b[k].x;
 88 }
